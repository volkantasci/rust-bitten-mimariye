// Özyinelemeli Fibonacci
fn fib(n) {
    if n < 2 { return n; }
    return fib(n - 1) + fib(n - 2);
}

let i = 0;
while i <= 10 {
    print(fib(i));
    i = i + 1;
}
print(fib(20));
